import asyncio from aiogram import Bot, Dispatcher, F from aiogram.filters import Command from aiogram.types import Message, CallbackQuery from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton TOKEN = "ТВОЙ_ТОКЕН" bot = Bot(token=TOKEN) dp = Dispatcher() players = {} def get_player(user_id): if user_id not in players: players[user_id] = {"coins": 0, "power": 1} return players[user_id] def menu(): return InlineKeyboardMarkup(inline_keyboard=[ [InlineKeyboardButton(text="🪙 КЛИК", callback_data="click")], [InlineKeyboardButton(text="⚡ Улучшение", callback_data="upgrade")], [InlineKeyboardButton(text="💰 Баланс", callback_data="balance")] ]) @dp.message(Command("start")) async def start(message: Message): get_player(message.from_user.id) await message.answer( "🎮 Добро пожаловать в КЛИКЕР!\nНажимай на кнопку!", reply_markup=menu() ) @dp.callback_query(F.data == "click") async def click(call: CallbackQuery): player = get_player(call.from_user.id) player["coins"] += player["power"] await call.answer("🪙 +{}".format(player["power"])) await call.message.edit_text( f"🎮 КЛИКЕР\n\n💰 Монеты: {player['coins']}", reply_markup=menu() ) @dp.callback_query(F.data == "balance") async def balance(call: CallbackQuery): player = get_player(call.from_user.id) await call.answer( f"💰 Баланс: {player['coins']} монет", show_alert=True ) @dp.callback_query(F.data == "upgrade") async def upgrade(call: CallbackQuery): player = get_player(call.from_user.id) price = player["power"] * 50 if player["coins"] >= price: player["coins"] -= price player["power"] += 1 await call.answer("⚡ Улучшение куплено!") await call.message.edit_text( f"🎮 КЛИКЕР\n\n" f"💰 Монеты: {player['coins']}\n" f"⚡ За клик: {player['power']}", reply_markup=menu() ) else: await call.answer( f"❌ Нужно {price} монет!", show_alert=True ) async def main(): print("Бот запущен!") await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
