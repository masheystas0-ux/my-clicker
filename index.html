import asyncio import random from aiogram import Bot, Dispatcher, types, F from aiogram.filters import Command from aiogram.utils.keyboard import InlineKeyboardBuilder BOT_TOKEN = "ВАШ_ТОКЕН_ОТ_BOTFATHER" bot = Bot(token=BOT_TOKEN) dp = Dispatcher() # Простая база данных в памяти (для реального проекта лучше использовать SQLite или PostgreSQL) users_db = {} # Настройка кейса: (Название, Редкость, Шанс в %, Примерная цена) NFT_POOL = [ {"name": "🐸 Pepe Punk #104", "rarity": "Обычный", "chance": 60, "price": 50}, {"name": "💎 TON Crystal #88", "rarity": "Редкий", "chance": 25, "price": 150}, {"name": "👑 Durov's Cap #07", "rarity": "Эпический", "chance": 12, "price": 500}, {"name": "🚀 Golden Rocket #01", "rarity": "Легендарный", "chance": 3, "price": 2500}, ] CASE_COST = 100 # Стоимость одной прокрутки def get_user(user_id: int): if user_id not in users_db: # Выдаем 500 монет новичку для теста users_db[user_id] = {"balance": 500, "inventory": []} return users_db[user_id] def main_menu(): builder = InlineKeyboardBuilder() builder.button(text=f"🎰 Крутить кейс ({CASE_COST} ⭐️)", callback_data="spin_case") builder.button(text="🎒 Мой инвентарь", callback_data="inventory") builder.button(text="💰 Пополнить (тест +500)", callback_data="add_funds") builder.adjust(1) return builder.as_markup() @dp.message(Command("start")) async def cmd_start(message: types.Message): user = get_user(message.from_user.id) await message.answer( f"👋 Добро пожаловать в NFT-рулетку!\n" f"Твой баланс: **{user['balance']} ⭐️**\n\n" f"Испытай удачу и выбей редкий NFT:", reply_markup=main_menu(), parse_mode="Markdown" ) @dp.callback_query(F.data == "spin_case") async def process_spin(callback: types.CallbackQuery): user = get_user(callback.from_user.id) if user["balance"] < CASE_COST: await callback.answer("❌ Недостаточно средств на балансе!", show_alert=True) return # Списываем стоимость прокрутки user["balance"] -= CASE_COST # Выбираем случайный NFT с учетом весов (шансов) weights = [item["chance"] for item in NFT_POOL] won_nft = random.choices(NFT_POOL, weights=weights, k=1)[0] # Добавляем в инвентарь user["inventory"].append(won_nft) await callback.message.edit_text( f"🎉 **Тебе выпал NFT!**\n\n" f"🎁 Предмет: **{won_nft['name']}**\n" f"✨ Редкость: {won_nft['rarity']}\n" f"💎 Ценность: {won_nft['price']} ⭐️\n\n" f"Остаток баланса: **{user['balance']} ⭐️**", reply_markup=main_menu(), parse_mode="Markdown" ) await callback.answer() @dp.callback_query(F.data == "inventory") async def show_inventory(callback: types.CallbackQuery): user = get_user(callback.from_user.id) inv = user["inventory"] if not inv: text = "🎒 Твой инвентарь пока пуст. Крути кейс!" else: items_text = "\n".join([f"• {item['name']} ({item['rarity']}) — {item['price']} ⭐️" for item in inv]) text = f"🎒 **Твои NFT ({len(inv)} шт.):**\n\n{items_text}" await callback.message.edit_text(text, reply_markup=main_menu(), parse_mode="Markdown") await callback.answer() @dp.callback_query(F.data == "add_funds") async def add_test_funds(callback: types.CallbackQuery): user = get_user(callback.from_user.id) user["balance"] += 500 await callback.answer("✅ Начислено 500 тестовых ⭐️!", show_alert=True) await callback.message.edit_text( f"Твой баланс пополнен!\nТекущий баланс: **{user['balance']} ⭐️**", reply_markup=main_menu(), parse_mode="Markdown" ) async def main(): await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
